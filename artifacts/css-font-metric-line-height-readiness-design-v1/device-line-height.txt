  84:     /// Returns the computed line-height for the font in a given computed values instance.
  85:     ///
  86:     /// If you pass down an element, then the used line-height is returned.
  87:     pub fn calc_line_height(
  88:         &self,
  89:         font: &crate::properties::style_structs::Font,
  90:         writing_mode: WritingMode,
  91:         element: Option<GeckoElement>,
  92:     ) -> NonNegativeLength {
  93:         let pres_context = self.pres_context();
  94:         let line_height = font.get_line_height();
  95:         let au = Au(unsafe {
  96:             bindings::Gecko_CalcLineHeight(
  97:                 line_height,
  98:                 pres_context.map_or(std::ptr::null(), |pc| pc),
  99:                 writing_mode.is_text_vertical(),
 100:                 &**font,
 101:                 element.map_or(std::ptr::null(), |e| e.0),
 102:             )
 103:         });
 104:         NonNegativeLength::new(au.to_f32_px())
 105:     }
