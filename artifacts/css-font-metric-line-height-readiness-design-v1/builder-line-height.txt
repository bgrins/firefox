2624:     pub fn calc_line_height(
2625:         &self,
2626:         device: &Device,
2627:         line_height_base: LineHeightBase,
2628:         writing_mode: WritingMode,
2629:     ) -> computed::NonNegativeLength {
2630:         use crate::computed_value_flags::ComputedValueFlags;

2632:             LineHeightBase::CurrentStyle => (
2633:                 self.get_font(),

2635:             ),
2636:             LineHeightBase::InheritedStyle => (
2637:                 self.get_parent_font(),

2639:             ),

2641:         let line_height = font.get_line_height();

2645:         let lh = device.calc_line_height(font, writing_mode, None);
2646:         if line_height_base == LineHeightBase::InheritedStyle {
2647:             // Apply our own zoom if our style source is the parent style.
2648:             computed::NonNegativeLength::new(self.effective_zoom_for_inheritance.zoom(lh.px()))
2649:         } else {
2650:             lh
2651:         }
2652:     }
